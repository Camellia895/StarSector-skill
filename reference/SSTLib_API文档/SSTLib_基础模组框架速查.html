🔌 三、ModPlugin 入口类

所有模组的入口点。继承 BaseModPlugin，按需重写生命周期方法。

生命周期方法速览

方法调用时机常用场景
onApplicationLoad()游戏启动后加载配置、注册全局数据
onEnabled(wasEnabledBefore)模组被启用时首次启用初始化
onNewGame()新游戏创建时生成星区内容
onNewGameAfterProcGen()程序化生成后在程序生成基础上修改
onNewGameAfterEconomyLoad()经济加载后市场相关初始化
onGameLoad(newGame)读档/新游戏后注册监听器、脚本 ✨
beforeGameSave()存档保存前准备持久化数据
afterGameSave()存档保存后恢复临时状态
onDevModeF8Reload()F8 热重载调试用
configureXStream(x)存档序列化配置注册自定义类别名

完整模板

package org.myname.mymod;

import com.fs.starfarer.api.BaseModPlugin;
import com.fs.starfarer.api.Global;
import com.fs.starfarer.api.GameState;
import java.util.concurrent.*;

public class MyModPlugin extends BaseModPlugin {

    private static final String MOD_ID = "MyMod";
    private ScheduledExecutorService scheduler;

    // 游戏启动时调用（最早的入口）
    @Override
    public void onApplicationLoad() throws Exception {
        super.onApplicationLoad();
        // 加载模组配置文件
        // Global.getSettings().loadJSON("data/config/my_config.json");

        // 若需在主菜单就生效，启动调度器等待游戏就绪
        startScheduler();

        Global.getLogger(MyModPlugin.class).info("MyMod 已加载");
    }

    // 新游戏创建时调用
    @Override
    public void onNewGame() {
        super.onNewGame();
        // 生成新游戏内容
    }

    // 程序化生成完成后
    @Override
    public void onNewGameAfterProcGen() {
        super.onNewGameAfterProcGen();
        // 在程序生成的星系基础上修改
    }

    // 经济系统加载后（此时市场可用）
    @Override
    public void onNewGameAfterEconomyLoad() {
        super.onNewGameAfterEconomyLoad();
        // 市场相关初始化
    }

    // 读档完成 / 新游戏创建完成 ✨ 最常用
    @Override
    public void onGameLoad(boolean newGame) {
        super.onGameLoad(newGame);

        // 注册 transient 监听器和脚本
        // Global.getSector().addTransientListener(new MyListener());
        // Global.getSector().addTransientScript(new MyScript());

        // 兜底：确保调度器已 shutdown
        if (scheduler != null) {
            scheduler.shutdownNow();
            scheduler = null;
        }

        Global.getLogger(MyModPlugin.class).info("MyMod 注册完成");
    }

    // 存档保存前
    @Override
    public void beforeGameSave() {
        super.beforeGameSave();
        // 准备持久化数据
    }

    // 配置 XStream 序列化（如需持久化自定义类）
    /*
    @Override
    public void configureXStream(XStream x) {
        super.configureXStream(x);
        x.alias("MyData", MyDataClass.class);
    }
    */

    // 调度器：等待游戏进入 TITLE/CAMPAIGN 状态后注册
    private void startScheduler() {
        if (scheduler != null) {
            scheduler.shutdownNow();
        }
        scheduler = Executors.newSingleThreadScheduledExecutor();
        scheduler.scheduleAtFixedRate(() -> {
            try {
                GameState state = Global.getCurrentState();
                if (state == GameState.TITLE
                        || state == GameState.CAMPAIGN) {
                    // 注册主菜单级别的监听器
                    scheduler.shutdown();
                }
            } catch (Exception e) {
                scheduler.shutdown();
            }
        }, 0, 100, TimeUnit.MILLISECONDS);
    }
}
💡 最佳实践：
  • 监听器和脚本用 transient（临时）方式注册，在 onGameLoad() 重新注册，便于增删模组
  • 包名推荐格式：org/作者名/模组名/
  • 不要在 onApplicationLoad() 中使用 Global.getSector()（此时为 null）

💡 MutableStat 三种修改方式：

⏱️ 五、每帧脚本 (EveryFrameScript)

战役中每帧执行的脚本，通过 Global.getSector().addScript() 注册。

代码模板

package org.myname.mymod.scripts;

import com.fs.starfarer.api.EveryFrameScript;
import com.fs.starfarer.api.Global;

public class MyEveryFrameScript implements EveryFrameScript {

    private float elapsed = 0f;
    private boolean done = false;
    private final float interval;

    public MyEveryFrameScript(float intervalSeconds) {
        this.interval = intervalSeconds;
    }

    // 脚本是否完成（返回 true 时自动移除）
    @Override
    public boolean isDone() {
        return done;
    }

    // 游戏暂停时是否仍执行
    @Override
    public boolean runWhilePaused() {
        return false;
    }

    // 每帧调用，amount 为经过的秒数
    @Override
    public void advance(float amount) {
        elapsed += amount;

        // 每隔 interval 秒执行一次
        if (elapsed >= interval) {
            elapsed = 0f;
            doPeriodicTask();
        }
    }

    private void doPeriodicTask() {
        // 定期执行的逻辑
        // Global.getSector().getPlayerFleet()...
    }

    // 手动结束脚本
    public void end() {
        done = true;
    }
}
💡 注册方式：
  • addScript() — 持久化，存入存档
  • addTransientScript() — 临时，不存档（推荐）
  • addScriptWithTimeout() — 带超时自动移除

package org.myname.mymod.shipsystems;

import com.fs.starfarer.api.impl.combat.BaseShipSystemScript;
import com.fs.starfarer.api.combat.MutableShipStatsAPI;
import com.fs.starfarer.api.combat.ShipAPI;
import com.fs.starfarer.api.plugins.ShipSystemStatsScript;

public class MyShipSystem extends BaseShipSystemScript {

    private static final String KEY = "my_system";

    // 应用效果
    @Override
    public void apply(
            MutableShipStatsAPI stats,
            String id,
            State state,
            float effectLevel) {

        if (effectLevel <= 0) return;

        // 根据 effectLevel (0~1) 线性应用效果
        stats.getMaxSpeed().modifyPercent(id, 50f * effectLevel);
        stats.getAcceleration().modifyPercent(id, 100f * effectLevel);
    }

    // 移除效果
    @Override
    public void unapply(
            MutableShipStatsAPI stats,
            String id) {

        stats.getMaxSpeed().unmodify(id);
        stats.getAcceleration().unmodify(id);
    }

    // 状态显示数据
    @Override
    public StatusData getStatusData(
            int index,
            State state,
            float effectLevel) {

        if (index == 0) {
            return new StatusData("机动提升", false);
        }
        return null;
    }

    // 可选：覆盖持续时间
    @Override
    public ShipSystemStatsScript.DurationResult getActiveOverride(
            ShipAPI ship) {
        return null; // 返回 null 使用默认值
    }
}

package org.myname.mymod.dialog;

import com.fs.starfarer.api.campaign.InteractionDialogAPI;
import com.fs.starfarer.api.campaign.InteractionDialogPlugin;
import com.fs.starfarer.api.campaign.VisualPanelAPI;
import com.fs.starfarer.api.campaign.rules.MemoryAPI;

public class MyDialogPlugin implements InteractionDialogPlugin {

    private InteractionDialogAPI dialog;
    private MemoryAPI memory;

    private enum OptionId {
        INIT,
        OPTION_1,
        OPTION_2,
        LEAVE
    }

    @Override
    public void init(InteractionDialogAPI dialog) {
        this.dialog = dialog;
        this.memory = dialog.getInteractionTarget().getMemory();

        // 显示初始文本
        dialog.getTextPanel().addPara("你好，旅行者！");

        // 添加选项
        dialog.getOptionPanel().addOption(
            "选项一", OptionId.OPTION_1);
        dialog.getOptionPanel().addOption(
            "选项二", OptionId.OPTION_2);
        dialog.getOptionPanel().addOption(
            "离开", OptionId.LEAVE);

        // 显示人物头像
        // dialog.getVisualPanel().showPersonVisual(person);

        // 显示舰船图像
        // dialog.getVisualPanel().showFleetInfo(title, null, fleet, null);
    }

    @Override
    public void optionSelected(
            String optionText,
            Object optionData) {

        OptionId option = (OptionId) optionData;

        switch (option) {
            case OPTION_1:
                dialog.getTextPanel().addPara("你选择了选项一。");
                // 重新添加选项
                dialog.getOptionPanel().clearOptions();
                dialog.getOptionPanel().addOption(
                    "离开", OptionId.LEAVE);
                break;

            case OPTION_2:
                dialog.getTextPanel().addPara("你选择了选项二。");
                dialog.getOptionPanel().clearOptions();
                dialog.getOptionPanel().addOption(
                    "离开", OptionId.LEAVE);
                break;

            case LEAVE:
                dialog.dismiss();
                break;

🔮 十三、反射工具类（绕过安全限制）

🔒 安全限制：游戏脚本类加载器在类加载时拦截 java.lang.reflect.Method / Field，任何代码中声明这些类型（变量、参数、返回值）都会抛出 SecurityException，即使从未调用。
❌ IMPL_LOOKUP 方案已弃用：获取 IMPL_LOOKUP 字段本身需要反射 Field 类型，同样被拦截。
✅ 正确方案：Class.forName + MethodHandles.lookup()，完全不引用反射类型。

ReflectionUtils 模板（Java 版，ChineseInputFix 实测可行）

核心思想：用 Class.forName() 获取反射类的 Class 对象，用 MethodHandles.lookup().findVirtual() 创建 MethodHandle，所有反射对象一律当作 Object 处理。

package org.myname.mymod.util;

import com.fs.starfarer.api.Global;
import java.lang.invoke.MethodHandle;
import java.lang.invoke.MethodHandles;
import java.lang.invoke.MethodType;
import java.util.*;

public class ReflectionUtils {

    // 反射类的 Class 对象（用 Class.forName 获取，不直接 import）
    private static final Class methodClass;
    private static final Class fieldClass;

    // MethodHandle 用于调用 Method/Field 的方法（避免直接引用类型）
    private static final MethodHandle invokeMethodHandle;
    private static final MethodHandle getMethodNameHandle;
    private static final MethodHandle getMethodParametersHandle;
    private static final MethodHandle setFieldAccessibleHandle;
    private static final MethodHandle getFieldHandle;
    private static final MethodHandle getFieldNameHandle;

    // 缓存：类 -> (方法签名 -> Method 对象)
    private static final Map<Class, Map<String, Object>> methodCache = new HashMap<>();
    private static final Map<Class, Map<String, Object>> fieldCache = new HashMap<>();

    static {
        try {
            // 1. 用 Class.forName 获取反射类的 Class 对象
            ClassLoader jdkLoader = Class.class.getClassLoader();
            methodClass = Class.forName("java.lang.reflect.Method", false, jdkLoader);
            fieldClass = Class.forName("java.lang.reflect.Field", false, jdkLoader);

            // 2. 用 MethodHandles.lookup().findVirtual() 创建 MethodHandle
            invokeMethodHandle = MethodHandles.lookup().findVirtual(methodClass, "invoke",
                    MethodType.methodType(Object.class, Object.class, Object[].class));
            getMethodNameHandle = MethodHandles.lookup().findVirtual(methodClass, "getName",
                    MethodType.methodType(String.class));
            getMethodParametersHandle = MethodHandles.lookup().findVirtual(methodClass, "getParameterTypes",
                    MethodType.methodType(Class[].class));

            setFieldAccessibleHandle = MethodHandles.lookup().findVirtual(fieldClass, "setAccessible",
                    MethodType.methodType(void.class, boolean.class));
            getFieldHandle = MethodHandles.lookup().findVirtual(fieldClass, "get",
                    MethodType.methodType(Object.class, Object.class));
            getFieldNameHandle = MethodHandles.lookup().findVirtual(fieldClass, "getName",
                    MethodType.methodType(String.class));
        } catch (Throwable e) {
            throw new RuntimeException("ReflectionUtils 初始化失败", e);
        }
    }

    // 3. 所有反射对象当作 Object 处理，通过 MethodHandle 调用
    public static Object invoke(Object obj, String methodName, Object... args) {
        if (obj == null || methodName == null) return null;
        try {
            Class clazz = obj.getClass();
            Object method = findMethod(clazz, methodName, args.length);
            if (method != null) {
                return invokeMethodHandle.invoke(method, obj, args);
            }
        } catch (Throwable e) {
        }
        return null;
    }

    public static Object get(Object obj, String fieldName) {
        if (obj == null || fieldName == null) return null;
        try {
            Class clazz = obj.getClass();
            Object field = findField(clazz, fieldName);
            if (field != null) {
                return getFieldHandle.invoke(field, obj);
            }
        } catch (Throwable e) {
        }
        return null;
    }

    // 4. 查找方法：返回 Object，不是 Method
    private static Object findMethod(Class clazz, String methodName, int paramCount) {
        Map<String, Object> cache = methodCache.computeIfAbsent(clazz, k -> new HashMap<>());
        String key = methodName + ":" + paramCount;

        return cache.computeIfAbsent(key, k -> {
            try {
                // getMethods() 返回 Object[]，不是 Method[]
                Set<Object> methods = new HashSet<>();
                methods.addAll(Arrays.asList((Object[]) clazz.getMethods()));
                methods.addAll(Arrays.asList((Object[]) clazz.getDeclaredMethods()));

                for (Object method : methods) {
                    String name = (String) getMethodNameHandle.invoke(method);
                    if (!name.equals(methodName)) continue;
                    Class[] params = (Class[]) getMethodParametersHandle.invoke(method);
                    if (params.length != paramCount) continue;
                    return method;  // 返回 Object
                }
            } catch (Throwable e) {
            }
            return null;
        });
    }

    // 5. 查找字段：返回 Object，不是 Field
    private static Object findField(Class clazz, String fieldName) {
        Map<String, Object> cache = fieldCache.computeIfAbsent(clazz, k -> new HashMap<>());
        return cache.computeIfAbsent(fieldName, k -> {
            try {
                for (Object field : (Object[]) clazz.getDeclaredFields()) {
                    String name = (String) getFieldNameHandle.invoke(field);
                    if (name.equals(fieldName)) {
                        setFieldAccessibleHandle.invoke(field, true);
                        return field;  // 返回 Object
                    }
                }
            } catch (Throwable e) {
            }
            return null;
        });
    }
}

❌ 常见错误写法（会触发 SecurityException）

// ❌ 直接 import 反射类
import java.lang.reflect.Method;  // 类加载即抛异常
import java.lang.reflect.Field;

// ❌ 声明 Method/Field 类型变量
Method m = clazz.getMethod("getCore");  // 触发异常
Method[] methods = clazz.getMethods();  // 触发异常

// ❌ 使用 IMPL_LOOKUP（获取字段本身需要反射 Field）
Field f = MethodHandles.Lookup.class.getDeclaredField("IMPL_LOOKUP");  // 触发异常

使用示例

// 1. 获取未公开的内部对象
Object core = ReflectionUtils.invoke(state, "getCore");

// 2. UI 遍历：预定义 getter 名称列表（不要枚举 getMethods()）
String[] entryMethods = {"getCore", "getCurrentDialog", "getCurrentPanel"};
for (String name : entryMethods) {
    Object result = ReflectionUtils.invoke(campaignUI, name);
    if (result != null) { /* 处理 */ }
}

// 3. 读取私有字段
Object value = ReflectionUtils.get(instance, "privateField");
⚠️ 注意事项：
  • 反射结果务必缓存，避免每帧重复反射
  • 优先使用公开 API，反射仅用于访问未公开的内部类
  • 游戏更新可能导致字段/方法名变化，需做容错
  • 反射调用比直接调用慢，避免高频使用
